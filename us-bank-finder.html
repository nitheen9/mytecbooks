async function loadBanks() { showLoading("🔄 Loading banks..."); hideError(); try { const response = await fetch( API + "?type=banks", { method: "POST", headers: { "Content-Type": "application/json" }, body: "{}" } ); const data = await response.json(); if (!response.ok) { throw new Error( data.error || "Unable to load banks." ); } const banks = Array.isArray(data.banks) ? data.banks : []; if (banks.length === 0) { throw new Error( "No banks were returned." ); } bankSelect.innerHTML = ""; const firstOption = document.createElement( "option" ); firstOption.value = ""; firstOption.textContent = "Select Bank"; bankSelect.appendChild( firstOption ); banks.sort( function (a, b) { return a.name.localeCompare( b.name ); } ); for (const bank of banks) { if ( !Array.isArray( bank.certs ) || bank.certs.length === 0 ) { continue; } const option = document.createElement( "option" ); option.value = bank.certs.join(","); option.textContent = bank.name; bankSelect.appendChild( option ); } bankSelect.disabled = false; hideLoading(); } catch (error) { console.error( "Bank error:", error ); hideLoading(); showError( "Unable to load U.S. banks. " + error.message ); } }
